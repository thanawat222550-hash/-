<script>
let musicPlaying = false, noClicks = 0;
const bgMusic = document.getElementById('bgMusic');
const playBtn = document.getElementById('playBtn');
const musicNote = document.getElementById('musicNote');

function toggleMusic(){
if(!musicPlaying){
bgMusic.volume = .35;
bgMusic.play().catch(e=>console.log('Auto-play blocked — กดเพื่อเปิดเพลง'));
playBtn.textContent = '🔇';
musicNote.style.display = 'block';
musicPlaying = true;
}else{
bgMusic.pause();
playBtn.textContent = '🎵';
musicNote.style.display = 'none';
musicPlaying = false;
}
}

function runAway(){
noClicks++;
const b = document.getElementById('no');
if(noClicks<=4){
b.style.position = 'absolute';
b.style.left = Math.random()*60+15+'%';
b.style.top = (100+Math.random()80)+'px';
b.style.transform = scale(${1-noClicks.15}); document.getElementById('yes').style.transform =scale(${1+noClicks*.12});
if(noClicks===4){b.style.display='none';document.getElementById('yes').style.transform='scale(1.8)'}
}else{b.style.display='none'}
}

function forgive(){
document.getElementById('reply').innerHTML = '<h2>เย้!!! 🥹❤️</h2><p>ขอบคุณที่ให้อภัยเรานะ<br>เราจะพยายามทำให้เธอยิ้มเยอะๆ เลย</p>';
hearts();
}

function hearts(){
for(let i=0;i<30;i++){
let h=document.createElement('div');
h.className='heartfall';
h.textContent=['❤️','💕','💗','💖','💓'][Math.floor(Math.random()*5)];
h.style.left=Math.random()*100+'vw';
h.style.fontSize=16+Math.random()*30+'px';
h.style.animationDuration=3+Math.random()*4+'s';
document.body.appendChild(h);
setTimeout(()=>h.remove(),7000);
}
}
